    UserId INT PRIMARY KEY IDENTITY(1,1),
    Username NVARCHAR(100) UNIQUE NOT NULL,
    Password NVARCHAR(100) NOT NULL,
    Role NVARCHAR(50) NOT NULL
)

-- =============================================
-- SUPPLIERS
-- =============================================
CREATE TABLE Suppliers
(
    SupplierId INT PRIMARY KEY IDENTITY(1,1),
    SupplierCode NVARCHAR(50) UNIQUE,
    SupplierName NVARCHAR(100) NOT NULL,
    Phone VARCHAR(15),
    Address NVARCHAR(255),
    CreatedAt DATETIME DEFAULT GETDATE()
)
GO

-- =============================================
-- PRODUCTS
-- =============================================
CREATE TABLE Products
(
    ProductId INT PRIMARY KEY IDENTITY(1,1),
    SKU NVARCHAR(50) UNIQUE,
    ProductName NVARCHAR(150) NOT NULL,
    Quantity INT DEFAULT 0 CHECK(Quantity>=0),
    Price DECIMAL(18,2) CHECK(Price>=0),
    MinStock INT DEFAULT 10,
    SupplierId INT,
    CreatedAt DATETIME DEFAULT GETDATE(),

    CONSTRAINT FK_Product_Supplier
    FOREIGN KEY(SupplierId)
    REFERENCES Suppliers(SupplierId)
)
GO

-- =============================================
-- STOCK LOGS
-- =============================================
CREATE TABLE StockLogs
(
    LogId INT PRIMARY KEY IDENTITY(1,1),
    ProductId INT NOT NULL,
    Quantity INT NOT NULL,
    Type NVARCHAR(20),
    Note NVARCHAR(255),
    CreatedAt DATETIME DEFAULT GETDATE(),

    CONSTRAINT FK_Stock_Product
    FOREIGN KEY(ProductId)
    REFERENCES Products(ProductId)
)
GO

-- =============================================
-- LOGIN
-- =============================================
CREATE OR ALTER PROCEDURE sp_Login
@Username NVARCHAR(100),
@Password NVARCHAR(100)
AS
BEGIN
    SELECT * FROM Users
    WHERE Username=@Username AND Password=@Password
END
GO

-- =============================================
-- SUPPLIER CRUD
-- =============================================
CREATE OR ALTER PROCEDURE sp_AddSupplier
@SupplierCode NVARCHAR(50),
@SupplierName NVARCHAR(100),
@Phone VARCHAR(15),
@Address NVARCHAR(255)
AS
INSERT INTO Suppliers
VALUES(@SupplierCode,@SupplierName,@Phone,@Address,GETDATE())
GO

CREATE OR ALTER PROCEDURE sp_GetSuppliers
AS
SELECT * FROM Suppliers
GO

CREATE OR ALTER PROCEDURE sp_UpdateSupplier
@SupplierId INT,
@SupplierCode NVARCHAR(50),
@SupplierName NVARCHAR(100),
@Phone VARCHAR(15),
@Address NVARCHAR(255)
AS
UPDATE Suppliers
SET SupplierCode=@SupplierCode,
SupplierName=@SupplierName,
Phone=@Phone,
Address=@Address
WHERE SupplierId=@SupplierId
GO

CREATE OR ALTER PROCEDURE sp_DeleteSupplier
@SupplierId INT
AS
DELETE FROM Suppliers WHERE SupplierId=@SupplierId
GO

-- =============================================
-- PRODUCT CRUD
-- =============================================
CREATE OR ALTER PROCEDURE sp_AddProduct
@SKU NVARCHAR(50),
@ProductName NVARCHAR(150),
@Quantity INT,
@Price DECIMAL(18,2),
@MinStock INT,
@SupplierId INT
AS
INSERT INTO Products
VALUES(@SKU,@ProductName,@Quantity,@Price,@MinStock,@SupplierId,GETDATE())
GO

CREATE OR ALTER PROCEDURE sp_GetProducts
AS
SELECT p.*,s.SupplierName
FROM Products p
LEFT JOIN Suppliers s ON p.SupplierId=s.SupplierId
GO

CREATE OR ALTER PROCEDURE sp_UpdateProduct
@ProductId INT,
@SKU NVARCHAR(50),
@ProductName NVARCHAR(150),
@Quantity INT,
@Price DECIMAL(18,2),
@MinStock INT,
@SupplierId INT
AS
UPDATE Products
SET SKU=@SKU,
ProductName=@ProductName,
Quantity=@Quantity,
Price=@Price,
MinStock=@MinStock,
SupplierId=@SupplierId
WHERE ProductId=@ProductId
GO

CREATE OR ALTER PROCEDURE sp_DeleteProduct
@ProductId INT
AS
DELETE FROM Products WHERE ProductId=@ProductId
GO

-- =============================================
-- IMPORT
-- =============================================
CREATE OR ALTER PROCEDURE sp_ImportStock
@ProductId INT,
@Quantity INT,
@Note NVARCHAR(255)
AS
BEGIN
UPDATE Products SET Quantity=Quantity+@Quantity WHERE ProductId=@ProductId

INSERT INTO StockLogs(ProductId,Quantity,Type,Note)
VALUES(@ProductId,@Quantity,'IMPORT',@Note)
END
GO

-- =============================================
-- EXPORT
-- =============================================
CREATE OR ALTER PROCEDURE sp_ExportStock
@ProductId INT,
@Quantity INT,
@Note NVARCHAR(255)
AS
BEGIN
DECLARE @CurrentStock INT

SELECT @CurrentStock=Quantity
FROM Products
WHERE ProductId=@ProductId

IF(@CurrentStock<@Quantity)
BEGIN
RAISERROR(N'Không đủ hàng',16,1)
RETURN
END

UPDATE Products
SET Quantity=Quantity-@Quantity
WHERE ProductId=@ProductId

INSERT INTO StockLogs(ProductId,Quantity,Type,Note)
VALUES(@ProductId,@Quantity,'EXPORT',@Note)
END
GO

-- =============================================
-- REPORTS
-- =============================================
CREATE OR ALTER PROCEDURE sp_LowStockReport
AS
SELECT * FROM Products
WHERE Quantity<=MinStock
GO

CREATE OR ALTER PROCEDURE sp_InventoryHistory
AS
SELECT sl.*,p.ProductName,p.SKU
FROM StockLogs sl
JOIN Products p ON sl.ProductId=p.ProductId
ORDER BY sl.CreatedAt DESC
GO

CREATE OR ALTER PROCEDURE sp_StockDashboard
AS
SELECT
COUNT(*) TotalProducts,
SUM(Quantity) TotalStock,
(SELECT COUNT(*) FROM Products WHERE Quantity<=MinStock) LowStockProducts
FROM Products
GO

CREATE OR ALTER PROCEDURE sp_SearchProducts
@Keyword NVARCHAR(100)
AS
SELECT *
FROM Products
WHERE ProductName LIKE '%'+@Keyword+'%'
OR SKU LIKE '%'+@Keyword+'%'
GO

CREATE OR ALTER PROCEDURE sp_GetProductsPaging
@Page INT,
@PageSize INT
AS
SELECT *
FROM Products
ORDER BY ProductId
OFFSET(@Page-1)*@PageSize ROWS
FETCH NEXT @PageSize ROWS ONLY
GO

CREATE OR ALTER PROCEDURE sp_StockReport
AS
SELECT
p.ProductName,
SUM(CASE WHEN sl.Type='IMPORT' THEN sl.Quantity ELSE 0 END) TotalImport,
SUM(CASE WHEN sl.Type='EXPORT' THEN sl.Quantity ELSE 0 END) TotalExport
FROM StockLogs sl
JOIN Products p ON sl.ProductId=p.ProductId
GROUP BY p.ProductName
GO
-- =============================================
-- ĐĂNG KÝ TÀI KHOẢN
-- =============================================
CREATE OR ALTER PROCEDURE sp_Register
(
    @Username NVARCHAR(100),
    @Password NVARCHAR(100),
    @Role INT
)
AS
BEGIN

    IF EXISTS
    (
        SELECT 1
        FROM Users
        WHERE Username = @Username
    )
    BEGIN
        RAISERROR(
            N'Tài khoản đã tồn tại',
            16,
            1
        )
        RETURN
    END

    INSERT INTO Users
    (
        Username,
        Password,
        Role
    )
    VALUES
    (
        @Username,
        @Password,
        @Role
    )

END
GO
